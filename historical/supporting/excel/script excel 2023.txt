  var endColumnLetter = "HE";   // End column letter
  
  var startColumn = columnLetterToIndex(startColumnLetter);
  var endColumn = columnLetterToIndex(endColumnLetter);
  var numCols = endColumn - startColumn + 1; // Calculate the number of columns to search through
  
  var numRows = sheet.getLastRow();
  var range = sheet.getRange(startRow, startColumn, numRows, numCols);
  var values = range.getValues();
  
  for (var row = 0; row < numRows; row++) {
    for (var col = 0; col < numCols; col++) {
      var cellValue = values[row][col];
      if (cellValue !== "") {
        for (var phrase in phraseMappings) {
          if (cellValue === phrase) {
            var targetCellAddress = phraseMappings[phrase];
            var targetCell = sheet.getRange(targetCellAddress);
            var sourceData = sheet.getRange(startRow + row, startColumn + col, 1, 7).getValues(); // Get the cell and the 6 cells to the right
            var targetColumn = targetCell.getColumn(); // Start copying at the target cell's column
            var targetRow = targetCell.getRow();
            var targetRange = sheet.getRange(targetRow, targetColumn, 1, 7);
            targetRange.setValues(sourceData); // Copy the source data to the target location
            sheet.getRange(startRow + row, startColumn + col).setValue(""); // Clear the original cell content
            break; // Exit the phrase loop after finding a match
          }
        }
      }
    }
  }
}

// Function to convert column letter to column index (A=1, B=2, ..., Z=26, AA=27, AB=28, ...)
function columnLetterToIndex(columnLetter) {
  var columnNumber = 0;
  var length = columnLetter.length;
  for (var i = 0; i < length; i++) {
    columnNumber += (columnLetter.charCodeAt(i) - 64) * Math.pow(26, length - i - 1);
  }
  return columnNumber;
}
