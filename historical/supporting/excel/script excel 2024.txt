// Function to convert column letter to column index (A=1, B=2, ..., Z=26, AA=27, AB=28, ...)
function columnLetterToIndex(columnLetter) {
  var columnNumber = 0;
  var length = columnLetter.length;
  for (var i = 0; i < length; i++) {
    columnNumber += (columnLetter.charCodeAt(i) - 64) * Math.pow(26, length - i - 1);
  }
  return columnNumber;
}
